# Price Tracker: the Python packages the app needs to run.
#
#     python -m venv .venv
#     .venv\Scripts\python.exe -m pip install -r requirements.txt
#
# Working on the code? Install requirements-dev.txt instead: it adds the tests.
#
# Each package is pinned from the version we've tested with up to its next
# major release, which is when breaking changes usually arrive.

# --- Reading the store pages ---------------------------------------------------
httpx>=0.28,<1.0                # downloads product pages (and talks to Telegram)
beautifulsoup4>=4.12,<5         # finds the price, title and photo in a page

# --- Keeping the price history -------------------------------------------------
SQLAlchemy>=2.1,<3              # the tables and queries, on PostgreSQL or SQLite
alembic>=1.20,<2                # creates the tables and upgrades them (migrations)
psycopg[binary]>=3.2,<4         # the PostgreSQL driver; "binary" ships its own libpq,
                                # so nothing needs compiling on Windows

# --- The web app ---------------------------------------------------------------
fastapi>=0.160,<1.0             # the API the dashboard talks to
uvicorn>=0.50,<1.0              # the web server that runs it
